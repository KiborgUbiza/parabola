```dataviewjs
let now = dv.date("today");
let start = now.startOf("year");
let end = now.endOf("year");

// получаем записи
let pages = dv.pages('"LIFE/Дневники"')
  .where(p => p.file.name >= start.toFormat("yyyy-MM-dd") 
           && p.file.name <= end.toFormat("yyyy-MM-dd"));

// считаем
let noSugar = pages.where(p => p.no_sugar === true).length;
let sugar = pages.where(p => p.no_sugar === false).length;
let total = noSugar + sugar;

// защита от деления на 0
let noSugarPercent = total ? (noSugar / total * 100).toFixed(1) : 0;
let sugarPercent = total ? (sugar / total * 100).toFixed(1) : 0;

// счётчик дней подряд без сахара
let streak = 0;
let sorted = pages.sort(p => p.file.name, 'desc');

for (let p of sorted) {
  if (p.no_sugar === true) streak++;
  else break;
}

// вывод
dv.paragraph(`🔥 Без сахара подряд: ${streak} дней`);
dv.paragraph(`📊 За год: ${noSugar} без сахара / ${sugar} с сахаром`);

// полоски
dv.el("div", `
<div style="margin-top:10px">
  <div>Без сахара (${noSugarPercent}%)</div>
  <div style="background:#ddd; border-radius:6px; overflow:hidden;">
    <div style="width:${noSugarPercent}%; background:#a8e6cf; height:20px;"></div>
  </div>

  <div style="margin-top:10px">С сахаром (${sugarPercent}%)</div>
  <div style="background:#ddd; border-radius:6px; overflow:hidden;">
    <div style="width:${sugarPercent}%; background:#ea5455; height:20px;"></div>
  </div>
</div>
`);
```
