```dataviewjs
let now = dv.date("today");

// месяц
let monthStart = now.startOf("month");
let monthEnd = now.endOf("month");

// год
let yearStart = now.startOf("year");
let yearEnd = now.endOf("year");

// все данные
let pages = dv.pages('"LIFE/Дневники"');

// функция группировки
function groupCount(data, start, end) {
  let filtered = data
    .where(p => p.file.name >= start.toFormat("yyyy-MM-dd") &&
                p.file.name <= end.toFormat("yyyy-MM-dd"))
    .groupBy(p => p.type);

  return Object.fromEntries(
    filtered.map(g => [g.key, g.rows.length])
  );
}

// считаем
let monthData = groupCount(pages, monthStart, monthEnd);
let yearData = groupCount(pages, yearStart, yearEnd);

// объединяем ключи
let types = new Set([
  ...Object.keys(monthData),
  ...Object.keys(yearData)
]);

// таблица
dv.table(
  ["Тип", "За месяц", "За год"],
  Array.from(types).map(t => [
    t,
    monthData[t] ?? 0,
    yearData[t] ?? 0
  ])
);
```

```dataviewjs
let now = dv.date("today");
let start = now.startOf("year");
let end = now.endOf("year");

// берём данные
let pages = dv.pages('"LIFE/Дневники"')
  .where(p => p.file.name >= start.toFormat("yyyy-MM-dd") 
           && p.file.name <= end.toFormat("yyyy-MM-dd"));

// счётчики
let noSugar = 0;   // 2
let medium = 0;    // 1
let sugar = 0;     // 0 или пусто

let total = pages.length;

// streak (2 подряд)
let streak = 0;
let sorted = pages.sort(p => p.file.name, 'desc');

for (let p of sorted) {
  let v = p.no_sugar ?? 0; // если нет значения → 0

  if (v === 2) streak++;
  else break;
}

// подсчёт
for (let p of pages) {
  let v = p.no_sugar ?? 0; // ❗ пусто = 0

  if (v === 2) noSugar++;
  else if (v === 1) medium++;
  else sugar++;
}

// проценты
let noSugarPercent = total ? (noSugar / total * 100).toFixed(1) : 0;
let mediumPercent = total ? (medium / total * 100).toFixed(1) : 0;
let sugarPercent = total ? (sugar / total * 100).toFixed(1) : 0;

// вывод
dv.paragraph(`🔥 Чистые дни подряд: ${streak}`);
dv.paragraph(`⌛ За год:
🥦 ${noSugar} | б сах
🥞 ${medium} | норм
🧊 ${sugar} | с сах`);

// полоски
dv.el("div", `
<div style="margin-top:10px">

  <div>б сах (${noSugarPercent}%)</div>
  <div style="background:#ddd; border-radius:6px; overflow:hidden;">
    <div style="width:${noSugarPercent}%; background:#a8e5f6; height:20px;"></div>
  </div>

  <div style="margin-top:10px">норм
  (${mediumPercent}%)</div>
  <div style="background:#ddd; border-radius:6px; overflow:hidden;">
    <div style="width:${mediumPercent}%; background:#dcf516; height:20px;"></div>
  </div>

  <div style="margin-top:10px">с сах (${sugarPercent}%)</div>
  <div style="background:#ddd; border-radius:6px; overflow:hidden;">
    <div style="width:${sugarPercent}%; background:#ea5455; height:20px;"></div>
  </div>

</div>
`);
```
