```dataviewjs
let now = dv.date("today");

// месяц
let monthStart = now.startOf("month");
let monthEnd = now.endOf("month");

// год
let yearStart = now.startOf("year");
let yearEnd = now.endOf("year");

// все данные
let pages = dv.pages('"LIFE/Дневники"');

// функция группировки
function groupCount(data, start, end) {
  let filtered = data
    .where(p => p.file.name >= start.toFormat("yyyy-MM-dd") &&
                p.file.name <= end.toFormat("yyyy-MM-dd"))
    .groupBy(p => p.type);

  return Object.fromEntries(
    filtered.map(g => [g.key, g.rows.length])
  );
}

// считаем
let monthData = groupCount(pages, monthStart, monthEnd);
let yearData = groupCount(pages, yearStart, yearEnd);

// объединяем ключи
let types = new Set([
  ...Object.keys(monthData),
  ...Object.keys(yearData)
]);

// таблица
dv.table(
  ["Тип", "За месяц", "За год"],
  Array.from(types).map(t => [
    t,
    monthData[t] ?? 0,
    yearData[t] ?? 0
  ])
);
```
